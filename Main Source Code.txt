//-----------------------------------------------
//Putty Communication protocol
//-----------------------------------------------
void SerCom0(unsigned char data)
  {
    while(!(SCI0SR1 & 0x80));
    SCI0DRL=data;  
  }
//-----------------------------------------------
//Putty initialization protocol
//-----------------------------------------------
void initialize(void){
    SCI0BDH = 0;
    SCI0BDL = 156;
    SCI0CR1 = 0;
    SCI0CR2 = 0x0C; 
  }
//-----------------------------------------------
//Increase clock frequency
//-----------------------------------------------
void set_clock_24mhz(void){
    SYNR = 0x02;
    REFDV = 0x00;
    PLLCTL = 0x60;
    while(!(0x08 & CRGFLG));
    CLKSEL |= 0x80;
  }
//-----------------------------------------------
//timer function
//-----------------------------------------------
void MSDelay(unsigned int itime) 
  {
    unsigned int i, j;
    for(i=0;i<itime;i++)
      for(j=0;j<24000;j++);
  }
//-----------------------------------------------
//DFT algorithm derived from source:
//https://www.geeksforgeeks.org/c/discrete-fourier-transform-and-its-inverse-using-c/
//Citation in report and presentation
//It was difficult to find a standard library written in pure C for code warrior
//It appeared that some of the C libraries were written in 2001 as well making compatibility a problem
//In the end, a simple DFT adapted for <math.h> worked the best
//-----------------------------------------------
void calculateDFT(void)
{
    k = 0;
    n = 0;
    i = 0;

    for (k = 0; k < 128; k++) {
        inputRe[k] = 0;
        inputIm[k] = 0;
        for (n = 0; n < 128; n++) {
            i = k * n;
            while(i > 127){
              i = i - 128;
            }
            inputRe[k] = inputRe[k] + values[n] * trigRe[i];
            inputIm[k] = inputIm[k] - values[n] * trigIm[i];
        }
        n = 0;
        i = 0;
        output[k] = (int)sqrt(inputRe[k] * inputRe[k] + inputIm[k] * inputIm[k]);
    }
    k = 0;
    n = 0;
    i = 0;
}
//-----------------------------------------------
//A trig function to be called for generating DFT sinusoidal portions 
//reference https://stackoverflow.com/questions/13466623/
//Citation in report and presentation
//-----------------------------------------------
void trig(void)
{
    k = 0;
    n = 0;
    
        for (n = 0; n < 128; n++) {
            trigRe[n] = (double)cos(2 * 3.141592 * (double)n * 0.0078125);
            trigIm[n] = (double)sin(2 * 3.141592 * (double)n * 0.0078125);
        }
        
    k = 0;
    n = 0;
}
//-----------------------------------------------
//Initializing the 7 segment displays
//-----------------------------------------------
void config7seg() 
{
DDRB = 0xFF;
DDRP_DDRP0 = 1;
DDRP_DDRP1 = 1;
DDRP_DDRP2 = 1;
DDRP_DDRP3 = 1;
}
